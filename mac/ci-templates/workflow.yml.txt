name: Agent Lab CI
on:
  pull_request:
  push:
    branches: [__BRANCH__]
  workflow_dispatch:
permissions:
  contents: read
jobs:
  writer-check:
    runs-on: ubuntu-latest
    timeout-minutes: 20
    steps:
      - uses: actions/checkout@v4
        with:
          ref: ${{ github.event.pull_request.head.sha || github.sha }}
          persist-credentials: false
      - uses: actions/setup-node@v4
        with:
          node-version: 22
          cache: npm
          cache-dependency-path: |
            package-lock.json
            backend/package-lock.json
      - run: npm ci
      - run: npm ci --prefix backend
      - run: npm run check
  browser-check:
    runs-on: ubuntu-latest
    timeout-minutes: 20
    steps:
      - uses: actions/checkout@v4
        with:
          ref: ${{ github.event.pull_request.head.sha || github.sha }}
          persist-credentials: false
      - uses: actions/setup-node@v4
        with:
          node-version: 22
          cache: npm
      - run: npm ci
      - run: npm install --no-save --package-lock=false --ignore-scripts @playwright/test@1.64.1
      - name: Cache pinned Chromium
        uses: actions/cache@v4
        with:
          path: ~/.cache/ms-playwright
          key: ${{ runner.os }}-${{ runner.arch }}-chromium-1.64.1-${{ hashFiles('package-lock.json') }}
      - name: Prepare Chromium with bounded retry
        timeout-minutes: 6
        env:
          PLAYWRIGHT_DOWNLOAD_CONNECTION_TIMEOUT: "30000"
        run: |
          for attempt in 1 2; do
            if timeout --kill-after=10s 150s npx --no-install playwright install --with-deps chromium; then
              exit 0
            fi
            echo "Chromium preparation attempt ${attempt} failed; retrying infrastructure only."
          done
          echo "::error title=CI infrastructure::Chromium preparation failed before browser tests."
          exit 1
      - run: npx --no-install playwright test --config playwright.agent-lab.config.mjs --reporter=list,html,./tests/agent-lab/evidence-reporter.cjs
      - uses: actions/upload-artifact@v4
        if: always()
        with:
          name: agent-lab-browser-${{ github.run_id }}-${{ github.run_attempt }}
          path: |
            playwright-report/
            test-results/
          retention-days: 14
